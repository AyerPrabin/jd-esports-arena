## A.1 Problem 1: Optimising Delivery Routes (Graph + Nearest Neighbour TSP)

```text
  22  # ==========================================================
  23  # PROBLEM 1: OPTIMISING DELIVERY ROUTES (GRAPH + NEAREST NEIGHBOUR)
  24  # ==========================================================
  25  class Location:
  26      def __init__(self, location_id, name, x, y):
  27          self.location_id = location_id
  28          self.name = name
  29          self.x = float(x)
  30          self.y = float(y)
  31  
  32      def __str__(self):
  33          return f"{self.name}({self.x:.0f},{self.y:.0f})"
  34  
  35  
  36  class DeliveryRouteOptimizer:
  37      """Locations form a complete graph (every pair connected by a straight-line
  38      distance). Nearest Neighbour is a greedy O(n^2) heuristic approximation
  39      of the Travelling Salesperson Problem.
  40      """
  41  
  42      def __init__(self):
  43          self.locations = {}  # location_id -> Location
  44  
  45      def add_location(self, location):
  46          self.locations[location.location_id] = location
  47          print(f"[GRAPH] Added node '{location.location_id}' -> {location}")
  48  
  49      def _distance(self, a, b):
  50          return ((a.x - b.x) ** 2 + (a.y - b.y) ** 2) ** 0.5
  51  
  52      def nearest_neighbour_route(self, start_id):
  53          if start_id not in self.locations:
  54              print(f"[ERROR] Start location '{start_id}' not found.")
  55              return None, 0.0
  56  
  57          unvisited = set(self.locations.keys())
  58          unvisited.remove(start_id)
  59          route = [start_id]
  60          current = start_id
  61          total_distance = 0.0
  62          comparisons = 0
  63  
  64          print(f"[NN-TSP] Starting route at '{start_id}'.")
  65          while unvisited:
  66              nearest_id = None
  67              nearest_dist = float("inf")
  68              for candidate_id in unvisited:
  69                  comparisons += 1
  70                  dist = self._distance(self.locations[current], self.locations[candidate_id])
  71                  print(f"   [COMPARE] {current} -> {candidate_id}: distance={dist:.2f}")
  72                  if dist < nearest_dist:
  73                      nearest_dist = dist
  74                      nearest_id = candidate_id
  75              print(f"   [CHOOSE] Nearest unvisited from '{current}' is '{nearest_id}' "
  76                    f"(distance={nearest_dist:.2f})")
  77              route.append(nearest_id)
  78              total_distance += nearest_dist
  79              unvisited.remove(nearest_id)
  80              current = nearest_id
  81  
  82          return_leg = self._distance(self.locations[current], self.locations[start_id])
  83          total_distance += return_leg
  84          route.append(start_id)
  85          print(f"   [RETURN] {current} -> {start_id}: distance={return_leg:.2f}")
  86  
  87          n = len(self.locations)
  88          print(f"[COMPLEXITY] Nearest Neighbour used {comparisons} distance comparisons for "
  89                f"n={n} nodes (upper bound n*(n-1)/2 = {n * (n - 1) // 2}) -> O(n^2).")
  90          return route, total_distance
  91  
  92      def brute_force_optimal(self, start_id, node_limit=8):
  93          """Exact TSP via exhaustive permutation search. O(n!) -- only safe for
  94          small n. Used to benchmark how close the heuristic gets to optimal.
  95          """
  96          import itertools
  97          n = len(self.locations)
  98          if n > node_limit:
  99              print(f"[BRUTE FORCE] Skipped: n={n} exceeds safe limit of {node_limit} (O(n!) blow-up).")
 100              return None, None
 101  
 102          others = [lid for lid in self.locations if lid != start_id]
 103          best_route, best_distance, tried = None, float("inf"), 0
 104          for perm in itertools.permutations(others):
 105              tried += 1
 106              route = [start_id] + list(perm) + [start_id]
 107              dist = sum(self._distance(self.locations[route[i]], self.locations[route[i + 1]])
 108                         for i in range(len(route) - 1))
 109              if dist < best_distance:
 110                  best_distance, best_route = dist, route
 111          print(f"[BRUTE FORCE] Evaluated {tried} permutations ((n-1)!) for the exact optimum -> O(n!).")
 112          return best_route, best_distance
```

## A.2 Problem 2: Dynamic Resource Allocation (Max-Heap + FCFS/SJF/Round Robin)

```text
 115  # ==========================================================
 116  # PROBLEM 2: DYNAMIC RESOURCE ALLOCATION (MAX-HEAP + FCFS/SJF/ROUND ROBIN)
 117  # ==========================================================
 118  class Process:
 119      def __init__(self, pid, name, priority, burst_time, cores_required, memory_required):
 120          self.pid = pid
 121          self.name = name
 122          self.priority = int(priority)
 123          self.burst_time = int(burst_time)          # ms of CPU time needed
 124          self.cores_required = int(cores_required)
 125          self.memory_required = int(memory_required)  # MB
 126  
 127      def __str__(self):
 128          return (f"{self.name}(PID{self.pid}, P{self.priority}, burst={self.burst_time}ms, "
 129                  f"cores={self.cores_required}, mem={self.memory_required}MB)")
 130  
 131  
 132  class ResourcePool:
 133      def __init__(self, total_cores, total_memory):
 134          self.total_cores = total_cores
 135          self.total_memory = total_memory
 136          self.available_cores = total_cores
 137          self.available_memory = total_memory
 138  
 139      def can_allocate(self, process):
 140          return (process.cores_required <= self.available_cores
 141                  and process.memory_required <= self.available_memory)
 142  
 143      def allocate(self, process):
 144          self.available_cores -= process.cores_required
 145          self.available_memory -= process.memory_required
 146  
 147      def release(self, process):
 148          self.available_cores += process.cores_required
 149          self.available_memory += process.memory_required
 150  
 151      def __str__(self):
 152          return f"Cores {self.available_cores}/{self.total_cores}, Memory {self.available_memory}/{self.total_memory}MB"
 153  
 154  
 155  class PriorityScheduler:
 156      """Binary max-heap stored in a flat Python list. Works on any object
 157      exposing a numeric `.priority` attribute -- reused here to order
 158      incoming Processes by priority before resources are allocated.
 159      Parent of index i is (i-1)//2. Children of index i are 2i+1, 2i+2.
 160      """
 161  
 162      def __init__(self):
 163          self.heap = []
 164  
 165      def insert(self, task):
 166          self.heap.append(task)
 167          index = len(self.heap) - 1
 168          print(f"[INSERT] Placed {task} at index {index} (bottom of heap).")
 169          self._heapify_up(index)
 170  
 171      def _heapify_up(self, index):
 172          parent_index = (index - 1) // 2
 173          while index > 0 and self.heap[parent_index].priority < self.heap[index].priority:
 174              child = self.heap[index]
 175              parent = self.heap[parent_index]
 176              print(f"   [HEAPIFY-UP] {child} at index {index} > parent {parent} "
 177                    f"at index {parent_index} -> SWAP")
 178              self.heap[index], self.heap[parent_index] = self.heap[parent_index], self.heap[index]
 179              index = parent_index
 180              parent_index = (index - 1) // 2
 181          if index == 0:
 182              print(f"   [HEAPIFY-UP] {self.heap[0]} has reached the ROOT (index 0). Bubble-up complete.")
 183          else:
 184              print(f"   [HEAPIFY-UP] Heap property satisfied at index {index}. Bubble-up complete.")
 185  
 186      def extract_max(self):
 187          if len(self.heap) == 0:
 188              print("[EXTRACT] Heap is empty. Nothing to extract.")
 189              return None
 190          if len(self.heap) == 1:
 191              removed = self.heap.pop()
 192              print(f"[EXTRACT] Removed last remaining item: {removed}")
 193              return removed
 194  
 195          max_item = self.heap[0]
 196          last_item = self.heap.pop()
 197          print(f"[EXTRACT] Removing root {max_item}. Moving last element {last_item} "
 198                f"from index {len(self.heap)} to root (index 0).")
 199          self.heap[0] = last_item
 200          self._heapify_down(0)
 201          return max_item
 202  
 203      def _heapify_down(self, index):
 204          size = len(self.heap)
 205          while True:
 206              largest = index
 207              left = 2 * index + 1
 208              right = 2 * index + 2
 209  
 210              if left < size and self.heap[left].priority > self.heap[largest].priority:
 211                  largest = left
 212              if right < size and self.heap[right].priority > self.heap[largest].priority:
 213                  largest = right
 214  
 215              if largest != index:
 216                  print(f"   [HEAPIFY-DOWN] {self.heap[largest]} at index {largest} > "
 217                        f"{self.heap[index]} at index {index} -> SWAP")
 218                  self.heap[index], self.heap[largest] = self.heap[largest], self.heap[index]
 219                  index = largest
 220              else:
 221                  print(f"   [HEAPIFY-DOWN] Heap property satisfied at index {index}. Sift-down complete.")
 222                  break
 223  
 224      def print_queue(self):
 225          print(f"Current queue: [{', '.join(str(t) for t in self.heap)}]")
 226  
 227  
 228  class TaskManager:
 229      """Assigns limited CPU/memory resources to incoming processes, ordered
 230      by priority (max-heap). Processes that arrive when resources are
 231      unavailable wait until something is released.
 232      """
 233  
 234      def __init__(self, total_cores, total_memory):
 235          self.pool = ResourcePool(total_cores, total_memory)
 236          self.ready_heap = PriorityScheduler()
 237          self.waiting = []
 238          self.running = []
 239          self._next_pid = 1
 240  
 241      def submit_process(self, name, priority, burst_time, cores_required, memory_required):
 242          process = Process(self._next_pid, name, priority, burst_time, cores_required, memory_required)
 243          self._next_pid += 1
 244          print(f"[SUBMIT] {process.name} (PID {process.pid}) queued with priority {process.priority}.")
 245          self.ready_heap.insert(process)
 246          return process
 247  
 248      def dispatch_all(self):
 249          print(f"[POOL] {self.pool}")
 250          while self.ready_heap.heap:
 251              process = self.ready_heap.extract_max()
 252              if self.pool.can_allocate(process):
 253                  self.pool.allocate(process)
 254                  self.running.append(process)
 255                  print(f"[ALLOCATE] {process.name} granted {process.cores_required} core(s), "
 256                        f"{process.memory_required}MB. Pool now: {self.pool}")
 257              else:
 258                  self.waiting.append(process)
 259                  print(f"[WAIT] Not enough resources for {process.name} "
 260                        f"(needs {process.cores_required} cores/{process.memory_required}MB, "
 261                        f"pool has {self.pool}). Added to waiting list.")
 262  
 263      def release_process(self, pid):
 264          for process in self.running:
 265              if process.pid == pid:
 266                  self.pool.release(process)
 267                  self.running.remove(process)
 268                  print(f"[RELEASE] {process.name} finished, resources returned. Pool now: {self.pool}")
 269                  self._retry_waiting()
 270                  return True
 271          print(f"[ERROR] PID '{pid}' is not currently running.")
 272          return False
 273  
 274      def _retry_waiting(self):
 275          still_waiting = []
 276          for process in self.waiting:
 277              if self.pool.can_allocate(process):
 278                  self.pool.allocate(process)
 279                  self.running.append(process)
 280                  print(f"[ALLOCATE] (from wait list) {process.name} granted resources. Pool now: {self.pool}")
 281              else:
 282                  still_waiting.append(process)
 283          self.waiting = still_waiting
 284  
 285      # ---- comparison scheduling algorithms (ordering only, no resource pool) ----
 286      @staticmethod
 287      def fcfs_schedule(processes):
 288          print("[FCFS] Execution order = arrival order:")
 289          clock = 0
 290          for p in processes:
 291              print(f"   t={clock:4d} -> {p.name} runs for {p.burst_time}ms")
 292              clock += p.burst_time
 293          print(f"[FCFS] Total turnaround time: {clock}ms")
 294          return clock
 295  
 296      @staticmethod
 297      def sjf_schedule(processes):
 298          """Non-preemptive Shortest-Job-First, ordered via manual selection sort."""
 299          remaining = list(processes)
 300          order = []
 301          while remaining:
 302              shortest = remaining[0]
 303              for p in remaining:
 304                  if p.burst_time < shortest.burst_time:
 305                      shortest = p
 306              order.append(shortest)
 307              remaining.remove(shortest)
 308  
 309          print("[SJF] Execution order (shortest burst first):")
 310          clock = 0
 311          for p in order:
 312              print(f"   t={clock:4d} -> {p.name} runs for {p.burst_time}ms")
 313              clock += p.burst_time
 314          print(f"[SJF] Total turnaround time: {clock}ms")
 315          return clock
 316  
 317      @staticmethod
 318      def round_robin_schedule(processes, quantum):
 319          print(f"[ROUND ROBIN] quantum={quantum}ms")
 320          queue = [[p, p.burst_time] for p in processes]
 321          clock = 0
 322          index = 0
 323          while queue:
 324              entry = queue[index]
 325              process, remaining = entry
 326              run_for = min(quantum, remaining)
 327              print(f"   t={clock:4d} -> {process.name} runs for {run_for}ms (remaining before: {remaining}ms)")
 328              clock += run_for
 329              remaining -= run_for
 330              if remaining <= 0:
 331                  print(f"      {process.name} COMPLETE at t={clock}")
 332                  queue.pop(index)
 333                  if queue:
 334                      index %= len(queue)
 335              else:
 336                  entry[1] = remaining
 337                  index = (index + 1) % len(queue)
 338          print(f"[ROUND ROBIN] Total time: {clock}ms")
 339          return clock
 340
```

```text
 343  # PROBLEM 3: SHARED RESOURCE BOOKING (SORTED ARRAY + BINARY SEARCH)
 344  # ==========================================================
 345  class User:
 346      def __init__(self, user_id, name):
 347          self.user_id = user_id
 348          self.name = name
 349  
 350      def __str__(self):

 357          self.name = name

 360          return self.name
 361  
 362  
 363  class Booking:
 364      def __init__(self, meeting_title, start_time, end_time, user=None, resource=None):
 365          self.title = meeting_title
 366          self.start = int(start_time)
 367          self.end = int(end_time)
 368          self.user = user
 369          self.resource = resource
 370  
 371      def __str__(self):
 372          who = f" by {self.user.name}" if self.user else ""
 373          return f"{self.title} ({self.start:04d} - {self.end:04d}){who}"
 374  
 375  
 376  class ResourceScheduler:
 377      """Bookings for ONE resource, kept in a chronologically sorted array.
 378      A hand-written binary search locates the O(log n) insertion index.
 379      """
 380  
 381      def __init__(self):
 382          self.schedule = []
 383  
 384      def _binary_search_insertion_index(self, new_booking):
 385          low = 0
 386          high = len(self.schedule)
 387          print(f"   [BINARY SEARCH] Locating insertion index for start={new_booking.start:04d} "
 388                f"across {len(self.schedule)} existing booking(s).")
 389  
 390          while low < high:
 391              mid = (low + high) // 2
 392              mid_start = self.schedule[mid].start
 393              if mid_start < new_booking.start:
 394                  print(f"      low={low}, high={high}, mid={mid} -> schedule[{mid}].start={mid_start:04d} "
 395                        f"< {new_booking.start:04d}, search RIGHT half")
 396                  low = mid + 1
 397              else:
 398                  print(f"      low={low}, high={high}, mid={mid} -> schedule[{mid}].start={mid_start:04d} "
 399                        f">= {new_booking.start:04d}, search LEFT half")
 400                  high = mid
 401  
 402          print(f"   [BINARY SEARCH] Converged: insertion index = {low}")
 403          return low
 404  
 405      def request_booking(self, new_booking):
 406          print(f"[REQUEST] {new_booking}")
 407          insertion_index = self._binary_search_insertion_index(new_booking)
 408  
 409          if insertion_index > 0 and self.schedule[insertion_index - 1].end > new_booking.start:
 410              print(f"   [BOUNDARY CHECK] Left neighbour {self.schedule[insertion_index - 1]} "
 411                    f"ends after {new_booking.start:04d}.")
 412              print(f"   -> [REJECTED] Conflict with previous booking.")
 413              return False
 414  
 415          if insertion_index < len(self.schedule) and self.schedule[insertion_index].start < new_booking.end:
 416              print(f"   [BOUNDARY CHECK] Right neighbour {self.schedule[insertion_index]} "
 417                    f"starts before {new_booking.end:04d}.")
 418              print(f"   -> [REJECTED] Conflict with upcoming booking.")
 419              return False
 420  
 421          print(f"   [BOUNDARY CHECK] No overlap with neighbouring bookings.")
 422          self.schedule.insert(insertion_index, new_booking)
 423          print(f"   -> [SUCCESS] {new_booking.title} inserted at index {insertion_index} "
 424                f"(array elements shifted right to make space).")
 425          return True
 426  
 427  
 428  class BookingSystem:
 429      """Manages many shared resources, each with its own sorted-array scheduler,
 430      so bookings on one resource never block another.
 431      """
 432  
 433      def __init__(self):
 434          self.resources = {}   # resource_id -> Resource
 435          self.schedulers = {}  # resource_id -> ResourceScheduler
 436  
 437      def add_resource(self, resource):
 438          self.resources[resource.resource_id] = resource
 439          self.schedulers[resource.resource_id] = ResourceScheduler()
 440          print(f"[RESOURCE] Registered shared resource '{resource}' (ID {resource.resource_id}).")
 441  
 442      def book(self, resource_id, user, title, start, end):
 443          if resource_id not in self.schedulers:
 444              print(f"[ERROR] Resource '{resource_id}' not found.")
 445              return False
 446          booking = Booking(title, start, end, user=user, resource=self.resources[resource_id])
 447          return self.schedulers[resource_id].request_booking(booking)
```

## A.4 Problem 4: Inventory Management (Hash Table)

```text
 450  # ==========================================================
 451  # PROBLEM 4: INVENTORY MANAGEMENT (HASH TABLE)
 452  # ==========================================================
 453  class InventoryItem:
 454      def __init__(self, item_id, name, current_stock, threshold):
 455          self.item_id = item_id
 456          self.name = name
 457          self.current_stock = int(current_stock)
 458          self.threshold = int(threshold)
 459  
 460      def update_stock(self, quantity_change):
 461          self.current_stock += quantity_change
 462  
 463      def needs_reorder(self):
 464          return self.current_stock <= self.threshold
 465  
 466      def __str__(self):
 467          return f"{self.name} (Stock: {self.current_stock}, Threshold: {self.threshold})"
 468  
 469  
 470  class InventoryManager:
 471      """Parts catalog keyed by item_id in a Python dict (hash table), giving
 472      O(1) average-case direct access for stock adjustments.
 473      """
 474  
 475      def __init__(self):
 476          self.catalog = {}
 477  
 478      def register_item(self, new_item):
 479          self.catalog[new_item.item_id] = new_item
 480          print(f"[REGISTER] {new_item.name} stored under hash key '{new_item.item_id}'.")
 481  
 482      def adjust_stock(self, target_id, quantity_change):
 483          print(f"[LOOKUP] Direct O(1) hash access for key '{target_id}'.")
 484          if target_id in self.catalog:
 485              target_item = self.catalog[target_id]
 486              before = target_item.current_stock
 487              target_item.update_stock(quantity_change)
 488              print(f"[TRANSACTION] {target_item.name}: {before} -> {target_item.current_stock} "
 489                    f"(change: {quantity_change:+d})")
 490              if target_item.needs_reorder():
 491                  print(f"   --> [ALERT] Critical stock level for {target_item.name}! "
 492                        f"({target_item.current_stock} <= threshold {target_item.threshold})")
 493          else:
 494              print(f"[ERROR] Item ID '{target_id}' not found.")
 495  
 496      def search_by_name(self, keyword):
 497          keyword_lower = keyword.lower()
 498          results = [item for item in self.catalog.values() if keyword_lower in item.name.lower()]
 499          print(f"[SEARCH] '{keyword}' matched {len(results)} item(s).")
 500          return results
 501  
 502      def sorted_by_stock(self):
 503          """Manual insertion sort, ascending by current_stock -- O(n^2), explicit
 504          for assessment purposes rather than relying on a built-in sort.
 505          """
 506          items = list(self.catalog.values())
 507          for i in range(1, len(items)):
 508              key_item = items[i]
 509              j = i - 1
 510              while j >= 0 and items[j].current_stock > key_item.current_stock:
 511                  items[j + 1] = items[j]
 512                  j -= 1
 513              items[j + 1] = key_item
 514          return items
 515  
 516      def auto_reorder(self, restock_amount=None):
 517          """Scan the catalog; any item at/below threshold is automatically restocked."""
 518          reordered = []
 519          for item in self.catalog.values():
 520              if item.needs_reorder():
 521                  amount = restock_amount if restock_amount else item.threshold * 3
 522                  print(f"[AUTO-REORDER] {item.name} at {item.current_stock} "
 523                        f"(<= threshold {item.threshold}). Placing reorder for {amount} units.")
 524                  item.update_stock(amount)
 525                  reordered.append(item)
 526          if not reordered:
 527              print("[AUTO-REORDER] No items require reordering.")
 528          return reordered
 529  
 530      def generate_report(self):
 531          print("=" * 52)
 532          print(f"{'ID':<8}{'Name':<18}{'Stock':>8}{'Threshold':>12}{'Status':>10}")
 533          print("-" * 52)
```

## A.5 Problem 5: Bookstore Recommendation Engine (Jaccard Similarity)

```text
 540  # ==========================================================
 541  # PROBLEM 5: BOOKSTORE RECOMMENDATION ENGINE (JACCARD SIMILARITY)
 542  # ==========================================================
 543  class Book:
 544      def __init__(self, book_id, title, genre):
 545          self.book_id = book_id
 546          self.title = title
 547          self.genre = genre
 548  
 549      def __str__(self):
 550          return f"{self.title} [{self.genre}]"
 551  
 552  
 553  class BookUser:
 554      def __init__(self, user_id, name):
 555          self.user_id = user_id
 556          self.name = name
 557  
 558      def __str__(self):
 559          return self.name
 560  
 561  
 562  class RecommendationEngine:
 563      """Collaborative filtering: users who bought similar books (by Jaccard
 564      similarity of their purchase sets) are used to recommend unread titles.
 565      """
 566  
 567      def __init__(self):
 568          self.books = {}       # book_id -> Book
 569          self.users = {}       # user_id -> BookUser
 570          self.purchases = {}   # user_id -> set(book_id)
 571  
 572      def add_book(self, book):
 573          self.books[book.book_id] = book
 574          print(f"[CATALOG] Added book '{book.book_id}' -> {book}")
 575  
 576      def add_user(self, user):
 577          self.users[user.user_id] = user
 578          self.purchases.setdefault(user.user_id, set())
 579          print(f"[USER] Registered '{user.user_id}' -> {user}")
 580  
 581      def record_purchase(self, user_id, book_id):
 582          self.purchases.setdefault(user_id, set()).add(book_id)
 583          buyer = self.users.get(user_id, user_id)
 584          book = self.books.get(book_id, book_id)
 585          print(f"[PURCHASE] {buyer} bought {book}")
 586  
 587      def jaccard_similarity(self, user_a, user_b):
 588          set_a = self.purchases.get(user_a, set())
 589          set_b = self.purchases.get(user_b, set())
 590          intersection = set_a & set_b
 591          union = set_a | set_b
 592          score = len(intersection) / len(union) if union else 0.0
 593          print(f"   [JACCARD] {user_a} vs {user_b}: |intersection|={len(intersection)}, "
 594                f"|union|={len(union)} -> similarity={score:.2f}")
 595          return score
 596  
 597      def find_similar_users(self, target_user, top_n=3):
 598          print(f"[SIMILARITY] Comparing '{target_user}' against all other users...")
 599          scores = []
 600          for other_id in self.purchases:
 601              if other_id == target_user:
 602                  continue
 603              scores.append((other_id, self.jaccard_similarity(target_user, other_id)))
 604          scores.sort(key=lambda pair: pair[1], reverse=True)
 605          u = len(scores)
 606          print(f"[COMPLEXITY] Compared against {u} user(s), each an O(b) set operation "
 607                f"-> O(u*b) for u users and b books/user.")
 608          return scores[:top_n]
 609  
 610      def recommend(self, target_user, top_n=3):
 611          already_owned = self.purchases.get(target_user, set())
 612          similar_users = self.find_similar_users(target_user)
 613          candidate_scores = {}
 614          for other_id, similarity in similar_users:
 615              if similarity <= 0:
 616                  continue
 617              for book_id in self.purchases.get(other_id, set()):
 618                  if book_id not in already_owned:
 619                      candidate_scores[book_id] = candidate_scores.get(book_id, 0) + similarity
 620  
 621          ranked = sorted(candidate_scores.items(), key=lambda pair: pair[1], reverse=True)[:top_n]
 622          recommendations = [self.books[book_id] for book_id, _ in ranked if book_id in self.books]
 623          print(f"[RECOMMEND] Top picks for '{target_user}': {[str(b) for b in recommendations]}")
 624          return recommendations
 625
```
